<script>

const levels = [

    // Level 1
    [
        "cat",
        "dog",
        "sun",
        "book",
        "tree",
        "home",
        "game",
        "school",
        "mouse",
        "water"
    ],

    // Level 2
    [
        "The sun is bright",
        "I love typing",
        "Practice makes perfect",
        "The cat is sleeping",
        "Learning is fun",
        "Keep your fingers moving"
    ],

    // Level 3
    [
        "Typing quickly requires regular practice",
        "Always look at the screen while typing",
        "Accuracy is more important than speed",
        "Keep your hands in the correct position",
        "Good typing skills save a lot of time"
    ],

    // Level 4
    [
        "The quick brown fox jumps over the lazy dog",
        "Technology is changing the way we communicate",
        "Programming requires patience practice and creativity",
        "A fast typist can write ideas without losing focus"
    ],

    // Level 5
    [
        "Success comes from consistent practice and determination",
        "The more accurately you type the faster you will become",
        "Developing good typing habits will improve your productivity",
        "Great programmers spend a lot of time communicating their ideas"
    ]

];

let level = 1;
let currentText = "";
let startTime = 0;
let timer = null;
let timeLeft = 30;
let score = 0;
let mistakes = 0;
let totalTyped = 0;
let started = false;

const textDisplay = document.getElementById("textDisplay");
const input = document.getElementById("input");

const levelDisplay = document.getElementById("level");
const timeDisplay = document.getElementById("time");
const wpmDisplay = document.getElementById("wpm");
const accuracyDisplay = document.getElementById("accuracy");
const scoreDisplay = document.getElementById("score");
const progress = document.getElementById("progress");
const message = document.getElementById("message");

const startBtn = document.getElementById("startBtn");
const restartBtn = document.getElementById("restartBtn");


// Get random text for current level
function getRandomText() {

    const words = levels[level - 1];

    return words[Math.floor(Math.random() * words.length)];
}


// Display text with character highlighting
function displayText() {

    textDisplay.innerHTML = "";

    for (let i = 0; i < currentText.length; i++) {

        const span = document.createElement("span");

        span.textContent = currentText[i];

        if (i < input.value.length) {

            if (input.value[i] === currentText[i]) {
                span.className = "correct";
            } else {
                span.className = "incorrect";
            }

        } else if (i === input.value.length) {

            span.className = "current";
        }

        textDisplay.appendChild(span);
    }

    updateProgress();
}


// Start game
function startGame() {

    level = 1;
    score = 0;
    mistakes = 0;
    totalTyped = 0;

    started = true;

    document.getElementById("gameArea").classList.remove("hidden");
    document.getElementById("gameOver").classList.add("hidden");

    input.disabled = false;
    input.value = "";

    timeLeft = 30;

    currentText = getRandomText();

    levelDisplay.textContent = level;
    scoreDisplay.textContent = score;
    timeDisplay.textContent = timeLeft;

    message.textContent = "Type the text as accurately as possible!";

    displayText();

    input.focus();

    startTime = Date.now();

    clearInterval(timer);

    timer = setInterval(updateTimer, 1000);
}


// Timer
function updateTimer() {

    timeLeft--;

    timeDisplay.textContent = timeLeft;

    updateWPM();

    if (timeLeft <= 0) {
        finishGame();
    }
}


// Typing event
input.addEventListener("input", function() {

    if (!started) return;

    const typed = input.value;

    totalTyped = typed.length;

    mistakes = 0;

    for (let i = 0; i < typed.length; i++) {

        if (typed[i] !== currentText[i]) {
            mistakes++;
        }
    }

    updateAccuracy();
    updateWPM();
    displayText();

    // Text completed
    if (typed === currentText) {

        completeText();
    }
});


// Complete current text
function completeText() {

    score += (currentText.length * level);

    scoreDisplay.textContent = score;

    message.textContent = "? Great! Next challenge...";

    // Level up after completing enough text
    const required = level * 2;

    if (!window.completedTexts) {
        window.completedTexts = 0;
    }

    window.completedTexts++;

    if (window.completedTexts >= required) {

        if (level < levels.length) {

            level++;

            window.completedTexts = 0;

            message.textContent =
                "?? Level Up! Welcome to Level " + level;

            levelDisplay.textContent = level;

        } else {

            finishGame();
            return;
        }
    }

    setTimeout(() => {

        input.value = "";

        currentText = getRandomText();

        displayText();

    }, 500);
}


// Accuracy
function updateAccuracy() {

    if (totalTyped === 0) {
        accuracyDisplay.textContent = "100%";
        return;
    }

    const accuracy =
        Math.max(
            0,
            Math.round(
                ((totalTyped - mistakes) / totalTyped) * 100
            )
        );

    accuracyDisplay.textContent = accuracy + "%";
}


// WPM calculation
function updateWPM() {

    if (!startTime) return;

    const elapsedMinutes =
        (Date.now() - startTime) / 60000;

    if (elapsedMinutes <= 0) return;

    const words = totalTyped / 5;

    const wpm = Math.round(words / elapsedMinutes);

    wpmDisplay.textContent = wpm;
}


// Progress bar
function updateProgress() {

    if (!currentText.length) return;

    const percentage =
        Math.min(
            100,
            (input.value.length / currentText.length) * 100
        );

    progress.style.width = percentage + "%";
}


// Finish game
function finishGame() {

    clearInterval(timer);

    started = false;

    input.disabled = true;

    updateWPM();

    const best =
        Math.max(
            score,
            Number(localStorage.getItem("typingBest") || 0)
        );

    localStorage.setItem("typingBest", best);

    document.getElementById("gameArea").classList.add("hidden");
    document.getElementById("gameOver").classList.remove("hidden");

    document.getElementById("finalScore").textContent = score;
    document.getElementById("bestScore").textContent = best;
}


// Restart
function restartGame() {

    clearInterval(timer);

    window.completedTexts = 0;

    document.getElementById("gameOver").classList.add("hidden");
    document.getElementById("gameArea").classList.remove("hidden");

    startGame();
}


startBtn.addEventListener("click", startGame);

restartBtn.addEventListener("click", restartGame);


// Allow Ctrl + Enter to restart
document.addEventListener("keydown", function(e) {

    if (e.ctrlKey && e.key === "Enter") {
        restartGame();
    }

});
